using APILogin.Models;
using Microsoft.EntityFrameworkCore;
namespace API_Consultas
{
	public class Program
	{
		public static void Main(string[] args)
		{
			// Constructor del builder: obtiene y prepara la configuración de la aplicación
			var builder = WebApplication.CreateBuilder(args);
			// NOTA IMPORTANTE: Configuración de Kestrel ahora en appsettings.json
			builder.Services.AddControllers();
			// Registra ArchivoContext con inyección de dependencias - permite usar DbContext en controladores
			builder.Services.AddDbContext<ArchivoContext>(options =>
				options.UseSqlServer(builder.Configuration.GetConnectionString("ConexionArchivo")));
			builder.Services.AddEndpointsApiExplorer();
			builder.Services.AddSwaggerGen();
			var app = builder.Build();
			// Habilita Swagger UI solo en desarrollo
			if (app.Environment.IsDevelopment())
			{
				app.UseSwagger();
				app.UseSwaggerUI(c =>
				{
					c.SwaggerEndpoint("/swagger/v1/swagger.json", "API v1");
					c.RoutePrefix = string.Empty;
				});
			}
			// COMENTADO: app.UseHttpsRedirection(); (evita rechazos de peticiones HTTP desde Android)
			app.UseAuthorization();
			app.MapControllers();
			// app.Run() lee la URL configurada en appsettings.json bajo Kestrel.Endpoints
			app.Run();
		}
	}
}

using System;
using System.Collections.Generic;
using Microsoft.EntityFrameworkCore;
namespace APILogin.Models;
public partial class ArchivoContext : DbContext
{
	public ArchivoContext()
	{
	}
	public ArchivoContext(DbContextOptions<ArchivoContext> options)
		: base(options)
	{
	}
	// Colecciones que representan las tablas principales
	public virtual DbSet<Estado> Estados { get; set; }
	public virtual DbSet<Persona> Personas { get; set; }
	public virtual DbSet<Usuario> Usuarios { get; set; }
	protected override void OnConfiguring(DbContextOptionsBuilder optionsBuilder)
	{
		// Solo configura la conexión si NO viene ya configurada desde Program.cs (inyección de dependencias)
		if (!optionsBuilder.IsConfigured)
		{
			optionsBuilder.UseSqlServer(@"Server=.\SQLEXPRESS;Database=ARCHIVO;Trusted_Connection=SSPI;MultipleActiveResultSets=true;Trust Server Certificate=true");
		}
	}
	protected override void OnModelCreating(ModelBuilder modelBuilder)
	{
		// Configuración de tabla ESTADO
		modelBuilder.Entity<Estado>(entity =>
		{
			entity.HasKey(e => e.IdEstado);
			entity.ToTable("ESTADO");
			entity.Property(e => e.IdEstado)
				.ValueGeneratedNever()
				.HasColumnName("ID_ESTADO");
			entity.Property(e => e.NomEstado)
				.HasMaxLength(20)
				.IsUnicode(false)
				.HasColumnName("NOM_ESTADO");
		});
		// Configuración de tabla PERSONA
		modelBuilder.Entity<Persona>(entity =>
		{
			entity.HasKey(e => e.IdPersona);
			entity.ToTable("PERSONA");
			entity.Property(e => e.IdPersona).HasColumnName("ID_PERSONA");
			entity.Property(e => e.ApePersona)
				.HasMaxLength(30)
				.IsUnicode(false)
				.HasColumnName("APE_PERSONA");
			entity.Property(e => e.Edad).HasColumnName("EDAD");
			entity.Property(e => e.NomPersona)
				.HasMaxLength(30)
				.IsUnicode(false)
				.HasColumnName("NOM_PERSONA");
		});
		// Configuración de tabla USUARIO con relaciones foráneas
		modelBuilder.Entity<Usuario>(entity =>
		{
			entity.HasKey(e => e.NomUsuario);
			entity.ToTable("USUARIO");
			entity.Property(e => e.NomUsuario)
				.HasMaxLength(10)
				.IsUnicode(false)
				.HasColumnName("NOM_USUARIO");
			// Mapeo de la columna CLAVE para las credenciales
			entity.Property(e => e.Clave)
				.HasMaxLength(50)
				.IsUnicode(false)
				.HasDefaultValue("123456")
				.HasColumnName("CLAVE");
			entity.Property(e => e.IdEstado).HasColumnName("ID_ESTADO");
			entity.Property(e => e.IdPersona).HasColumnName("ID_PERSONA");
			// Relación con ESTADO: un usuario tiene un estado
			entity.HasOne(d => d.IdEstadoNavigation).WithMany(p => p.Usuarios)
				.HasForeignKey(d => d.IdEstado)
				.OnDelete(DeleteBehavior.ClientSetNull)
				.HasConstraintName("FK_USUARIO_ESTADO");
			// Relación con PERSONA: un usuario está asociado a una persona
			entity.HasOne(d => d.IdPersonaNavigation).WithMany(p => p.Usuarios)
				.HasForeignKey(d => d.IdPersona)
				.OnDelete(DeleteBehavior.ClientSetNull)
				.HasConstraintName("FK_USUARIO_PERSONA");
		});
		OnModelCreatingPartial(modelBuilder);
	}
	partial void OnModelCreatingPartial(ModelBuilder modelBuilder);
}
