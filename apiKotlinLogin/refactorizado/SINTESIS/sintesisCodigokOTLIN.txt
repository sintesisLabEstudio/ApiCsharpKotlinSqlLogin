		enableEdgeToEdge()
		setContentView(R.layout.activity_login)
		ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->

			v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
			insets

		val txtUsuario = findViewById<EditText>(R.id.txtNomUsuarioLogin)
		val txtClave = findViewById<EditText>(R.id.txtClaveLogin)
		val btnIngresar = findViewById<Button>(R.id.btnIngresarLogin)
		val btnVolver = findViewById<Button>(R.id.btnVolverLogin)
		btnVolver?.setOnClickListener {
			startActivity(Intent(this, MainActivity::class.java))
			finish()
		}
		btnIngresar?.setOnClickListener {
			val user = txtUsuario?.text.toString().trim()
			val pass = txtClave?.text.toString().trim()
			if (user.isEmpty() || pass.isEmpty()) {
				Toast.makeText(this, "Debe ingresar usuario y contraseña", Toast.LENGTH_SHORT).show()
			} else {
				ejecutarLogin(user, pass)
			}
		}
	}
	private fun ejecutarLogin(usuario: String, clave: String) {
		CoroutineScope(Dispatchers.IO).launch {
			try {
				val dtoUsuario = Usuario(usuario, clave, 0, 1)
				val llamada = RetrofitCliente.ruta.Login(dtoUsuario)
				withContext(Dispatchers.Main) {
					if (llamada.isSuccessful) {
						val body: Datos? = llamada.body()
						if (body?.respuesta == "OK") {
							Toast.makeText(this@Login, "Bienvenido: ${body.mensaje}", Toast.LENGTH_SHORT).show()
							startActivity(Intent(this@Login, MainActivity::class.java))
							finish()
						} else {
							Toast.makeText(this@Login, body?.mensaje ?: "Credenciales inválidas", Toast.LENGTH_LONG).show()
						}
					} else {
						Toast.makeText(this@Login, "Error en el servidor: ${llamada.code()}", Toast.LENGTH_SHORT).show()
					}
				}
			} catch (ex: Exception) {
				withContext(Dispatchers.Main) {
					Toast.makeText(this@Login, "Error de red: ${ex.message}", Toast.LENGTH_LONG).show()
				}
			}
		}
	}
}

		listview.setOnItemClickListener { parent, _, position, _ ->

	fun cargaDatos(listview: ListView) {
		var adaptador: ArrayAdapter<*>
		val arregloListado = ArrayList<String>()
		CoroutineScope(Dispatchers.IO).launch {
			try {
				val llamada = RetrofitCliente.ruta.ListadoPersonas()
				if (llamada.isSuccessful) {
					val respuesta: Datos? = llamada.body()
					withContext(Dispatchers.Main) {
						val l = respuesta?.lista
						l?.forEach { item ->
							val id = item.idPersona
							val nombre = item.nomPersona
							val apellido = item.apePersona
							val edad = item.edad
							arregloListado.add("ID:$id - Nombre:$nombre  - Apellido:$apellido\nEdad:$edad")
						}
						adaptador = ArrayAdapter(this@ListadoPersonas, android.R.layout.simple_list_item_1, arregloListado)
						listview.adapter = adaptador
					}
				}
			} catch (ex: Exception) {
				withContext(Dispatchers.Main) {
					Toast.makeText(this@ListadoPersonas, "Error al cargar la lista: ${ex.message}", Toast.LENGTH_LONG).show()
				}
			}
		}
	}
}

			insets
		}
		val btnVolver = findViewById<Button>(R.id.btnLUVolver)
		val listview = findViewById<ListView>(R.id.lvListadoUsuarios)
		btnVolver.setOnClickListener {
			startActivity(Intent(this, MainActivity::class.java))
			finish()
		}
		cargaDatos(listview)
		listview.setOnItemClickListener { parent, _, position, _ ->
			val elemento = parent.getItemAtPosition(position)
			Toast.makeText(this, "Elemento: $elemento", Toast.LENGTH_SHORT).show()
		}
	}
	fun cargaDatos(listview: ListView) {
		var adaptador: ArrayAdapter<*>
		val arregloListado = ArrayList<String>()
		CoroutineScope(Dispatchers.IO).launch {
			try {
				val llamada = RetrofitCliente.ruta.ListadoUsuario()
				if (llamada.isSuccessful) {
					val respuesta: DatosUsuario? = llamada.body()
					withContext(Dispatchers.Main) {
						val l = respuesta?.lista
						l?.forEach { item ->
							val usuario = item.nomUsuario
							val nombre = item.nombreApellido
							val edad = item.edad
							val estado = item.nom_estado
							arregloListado.add("USUARIO: $usuario - Nombre: $nombre Edad: $edad \nEstado: $estado")
						}
						adaptador = ArrayAdapter(this@ListadoUsuarios, android.R.layout.simple_list_item_1, arregloListado)
						listview.adapter = adaptador
					}
				}
			} catch (ex: Exception) {
				withContext(Dispatchers.Main) {
					Toast.makeText(this@ListadoUsuarios, "Error al cargar usuarios: ${ex.message}", Toast.LENGTH_LONG).show()
				}
			}
		}
	}
}
