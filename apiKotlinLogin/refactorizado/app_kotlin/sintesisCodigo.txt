class IngresarUsuario : AppCompatActivity() {
	private var listaPersonasCache: List<Persona> = emptyList()
	override fun onCreate(savedInstanceState: Bundle?) {
		super.onCreate(savedInstanceState)
		enableEdgeToEdge()
		setContentView(R.layout.activity_ingresar_usuario)
		ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
			val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
			v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
			insets
		}
		val btnVolver = findViewById<Button>(R.id.btnIUVolver)
		val btnGuardar = findViewById<Button>(R.id.btnGuardarUsuario)
		val spinnerPersonas = findViewById<Spinner>(R.id.spinnerPersonasIngresarUsr)
		val txtNomUsuario = findViewById<EditText>(R.id.txtNomUsuario)
		val txtClave = findViewById<EditText>(R.id.txtClave)
		val txtIdPersona = findViewById<EditText>(R.id.txtIdPersona)
		val txtIdEstado = findViewById<EditText>(R.id.txtIdEstado)
		cargarSpinnerPersonas(spinnerPersonas, txtIdPersona)
		btnVolver.setOnClickListener { startActivity(Intent(this, MainActivity::class.java)); finish() }
		btnGuardar.setOnClickListener {
			val usr = txtNomUsuario.text.toString().trim()
			val pass = txtClave.text.toString().trim()
			val idPer = txtIdPersona.text.toString().trim()
			val idEst = txtIdEstado.text.toString().trim()
			if (usr.isEmpty() || idPer.isEmpty() || idEst.isEmpty()) { Toast.makeText(this, "Complete los campos obligatorios", Toast.LENGTH_SHORT).show() } else { guardarUsuario(usr, pass, idPer.toInt(), idEst.toInt()) }
		}
	}
	private fun cargarSpinnerPersonas(spinner: Spinner, txtIdPer: EditText) {
		CoroutineScope(Dispatchers.IO).launch {
			try {
				val llamada = RetrofitCliente.ruta.ListadoPersonas()
				if (llamada.isSuccessful) {
					val body = llamada.body()
					listaPersonasCache = body?.lista ?: emptyList()
					val nombresList = listaPersonasCache.map { "${it.idPersona} - ${it.nomPersona} ${it.apePersona}" }
					withContext(Dispatchers.Main) {
						val adapter = ArrayAdapter(this@IngresarUsuario, android.R.layout.simple_spinner_item, nombresList)
						adapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item)
						spinner.adapter = adapter
						spinner.onItemSelectedListener = object : AdapterView.OnItemSelectedListener {
							override fun onItemSelected(parent: AdapterView<*>, view: View?, position: Int, id: Long) {
								if (position < listaPersonasCache.size) {
									val personaSel = listaPersonasCache[position]
									txtIdPer.setText(personaSel.idPersona.toString())
								}
							}
							override fun onNothingSelected(parent: AdapterView<*>) {}
						}
					}
				}
			} catch (ex: Exception) { withContext(Dispatchers.Main) { Toast.makeText(this@IngresarUsuario, "Error al cargar personas: ${ex.message}", Toast.LENGTH_SHORT).show() } }
		}
	}
	private fun guardarUsuario(usuario: String, clave: String, idPersona: Int, idEstado: Int) {
		CoroutineScope(Dispatchers.IO).launch {
			try {
				val nuevoUsr = Usuario(usuario, if (clave.isEmpty()) "123456" else clave, idPersona, idEstado)
				val llamada = RetrofitCliente.ruta.InsertarUsuario(nuevoUsr)
				withContext(Dispatchers.Main) {
					if (llamada.isSuccessful) {
						val body: Datos? = llamada.body()
						Toast.makeText(this@IngresarUsuario, "${body?.respuesta}: ${body?.mensaje}", Toast.LENGTH_LONG).show()
					} else {
						Toast.makeText(this@IngresarUsuario, "Error del servidor: ${llamada.code()}", Toast.LENGTH_SHORT).show()
					}
				}
			} catch (ex: Exception) { withContext(Dispatchers.Main) { Toast.makeText(this@IngresarUsuario, "Error de conexión: ${ex.message}", Toast.LENGTH_LONG).show() } }
		}
	}
}

			v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
			insets
		}
		val btnVolver = findViewById<Button>(R.id.btnVolverEditUsuario)
		val btnEditar = findViewById<Button>(R.id.btnEditarUsuario)
		val btnEliminar = findViewById<Button>(R.id.btnEliminarUsuario)
		val btnLimpiar = findViewById<Button>(R.id.btnLimpiarEditUsuario)
		val spinnerPersonas = findViewById<Spinner>(R.id.spinnerPersonasUsrEdit)
		val spinnerUsuarios = findViewById<Spinner>(R.id.spinnerUsuariosUsrEdit)

		val txtClave = findViewById<EditText>(R.id.txtClaveEdit)
		val txtIdPersona = findViewById<EditText>(R.id.txtIdPersonaUsrEdit)

		btnEditar.setOnClickListener {

			val pass = txtClave.text.toString().trim()
			val idPerStr = txtIdPersona.text.toString().trim()
			val idEstStr = txtIdEstado.text.toString().trim()
			if (usr.isEmpty() || idPerStr.isEmpty() || idEstStr.isEmpty()) { Toast.makeText(this, "Debe ingresar usuario, ID persona e ID estado", Toast.LENGTH_SHORT).show() } else { val oldUsr = if (usuarioOriginalSeleccionado.isEmpty()) usr else usuarioOriginalSeleccionado; val claveFinal = if (pass.isEmpty()) "123456" else pass; ejecutarEditarUsuario(usr, claveFinal, idPerStr.toInt(), idEstStr.toInt(), oldUsr, spinnerPersonas, spinnerUsuarios, txtNomUsuario, txtClave, txtIdPersona, txtIdEstado) }

		btnEliminar.setOnClickListener {
			val usr = txtNomUsuario.text.toString().trim()

		}
	}
	private fun cargarSpinners(spinnerPer: Spinner, spinnerUsr: Spinner, txtUsr: EditText, txtPass: EditText, txtPer: EditText, txtEst: EditText) {
		CoroutineScope(Dispatchers.IO).launch {
			try {
				val llamadaPer = RetrofitCliente.ruta.ListadoPersonas()
				val llamadaUsr = RetrofitCliente.ruta.ListadoUsuario()
				if (llamadaPer.isSuccessful && llamadaUsr.isSuccessful) {
					listaPersonasCache = llamadaPer.body()?.lista ?: emptyList()
					listaUsuariosCache = llamadaUsr.body()?.lista ?: emptyList()
					val nombresPerList = listaPersonasCache.map { "${it.idPersona} - ${it.nomPersona} ${it.apePersona}" }
					withContext(Dispatchers.Main) {
						val adapterPer = ArrayAdapter(this@EditarUsuario, android.R.layout.simple_spinner_item, nombresPerList)
						adapterPer.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item)
						spinnerPer.adapter = adapterPer
						spinnerPer.onItemSelectedListener = object : AdapterView.OnItemSelectedListener {
							override fun onItemSelected(parent: AdapterView<*>, view: View?, position: Int, id: Long) {
								if (position < listaPersonasCache.size) {
									val personaSel = listaPersonasCache[position]
									txtPer.setText(personaSel.idPersona.toString())
									val nombreCompleto = "${personaSel.nomPersona} ${personaSel.apePersona}".trim()
									listaUsuariosFiltrados = listaUsuariosCache.filter { it.nombreApellido.equals(nombreCompleto, ignoreCase = true) }
									val nombresUsrList = listaUsuariosFiltrados.map { it.nomUsuario }
									val adapterUsr = ArrayAdapter(this@EditarUsuario, android.R.layout.simple_spinner_item, nombresUsrList)
									adapterUsr.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item)
									spinnerUsr.adapter = adapterUsr
									if (listaUsuariosFiltrados.isNotEmpty()) {
										spinnerUsr.setSelection(0)
										val usrSel = listaUsuariosFiltrados[0]
										usuarioOriginalSeleccionado = usrSel.nomUsuario
										txtUsr.setText(usrSel.nomUsuario)
										txtEst.setText(if (usrSel.nom_estado.equals("HABILITADO", ignoreCase = true)) "1" else "2")
									} else {
										usuarioOriginalSeleccionado = ""
										txtUsr.text.clear()
										txtPass.text.clear()
										txtEst.text.clear()
									}
									spinnerUsr.onItemSelectedListener = object : AdapterView.OnItemSelectedListener {
										override fun onItemSelected(parentUsr: AdapterView<*>, viewUsr: View?, posUsr: Int, idUsr: Long) {
											if (posUsr < listaUsuariosFiltrados.size) {
												val usuarioSel = listaUsuariosFiltrados[posUsr]
												usuarioOriginalSeleccionado = usuarioSel.nomUsuario
												txtUsr.setText(usuarioSel.nomUsuario)
												txtEst.setText(if (usuarioSel.nom_estado.equals("HABILITADO", ignoreCase = true)) "1" else "2")
											}
										}
										override fun onNothingSelected(parentUsr: AdapterView<*>) {}
									}
								}
							}
							override fun onNothingSelected(parent: AdapterView<*>) {}
						}
					}
				}
			} catch (ex: Exception) { withContext(Dispatchers.Main) { Toast.makeText(this@EditarUsuario, "Error al cargar datos: ${ex.message}", Toast.LENGTH_SHORT).show() } }
		}
	}
	private fun ejecutarEditarUsuario(usuario: String, clave: String, idPersona: Int, idEstado: Int, oldNomUsuario: String, spinnerPer: Spinner, spinnerUsr: Spinner, txtUsr: EditText, txtPass: EditText, txtPer: EditText, txtEst: EditText) {
		CoroutineScope(Dispatchers.IO).launch {
			try {
				val dto = Usuario(usuario, clave, idPersona, idEstado)
				val respuesta = RetrofitCliente.ruta.EditarUsuario(dto, oldNomUsuario)
				withContext(Dispatchers.Main) {
					if (respuesta.isSuccessful) { Toast.makeText(this@EditarUsuario, "${respuesta.body()?.respuesta}: ${respuesta.body()?.mensaje}", Toast.LENGTH_LONG).show(); cargarSpinners(spinnerPer, spinnerUsr, txtUsr, txtPass, txtPer, txtEst) } else { Toast.makeText(this@EditarUsuario, "Error del servidor: ${respuesta.code()}", Toast.LENGTH_SHORT).show() }
				}
			} catch (ex: Exception) { withContext(Dispatchers.Main) { Toast.makeText(this@EditarUsuario, "Error de red: ${ex.message}", Toast.LENGTH_LONG).show() } }
		}
	}
	private fun ejecutarEliminarUsuario(usuario: String, spinnerPer: Spinner, spinnerUsr: Spinner, txtUsr: EditText, txtPass: EditText, txtPer: EditText, txtEst: EditText) {
		CoroutineScope(Dispatchers.IO).launch {
			try {
				val dto = Usuario(usuario, "", 0, 0)
				val respuesta = RetrofitCliente.ruta.EliminarUsuario(dto)
				withContext(Dispatchers.Main) {
					if (respuesta.isSuccessful) { Toast.makeText(this@EditarUsuario, "${respuesta.body()?.respuesta}: ${respuesta.body()?.mensaje}", Toast.LENGTH_LONG).show(); txtUsr.text.clear(); txtPass.text.clear(); cargarSpinners(spinnerPer, spinnerUsr, txtUsr, txtPass, txtPer, txtEst) } else { Toast.makeText(this@EditarUsuario, "Error del servidor: ${respuesta.code()}", Toast.LENGTH_SHORT).show() }
				}
			} catch (ex: Exception) { withContext(Dispatchers.Main) { Toast.makeText(this@EditarUsuario, "Error de red: ${ex.message}", Toast.LENGTH_LONG).show() } }
		}
	}
